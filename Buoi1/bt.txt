module bai_tap_1 (
    input  [17:0] SW,
    output [17:0] LEDR
);
    assign LEDR = SW;
endmodule

module bai_tap_2 (
    input  [3:0] KEY,
    output [3:0] LEDG
);
    // Nếu nút nhấn tích cực mức thấp (nhấn = 0), dùng dấu ~ để đảo
    assign LEDG = ~KEY; 
endmodule

module bai_tap_3 (
    input  [1:0] SW,
    output reg [7:0] LEDR
);
    always @(*) begin
        case (SW)
            2'b00: LEDR = 8'b01010101; // LED 0, 2, 4, 6 sáng
            2'b01: LEDR = 8'b10101010; // LED 1, 3, 5, 7 sáng
            2'b10: LEDR = 8'b11111111; // Tất cả sáng
            2'b11: LEDR = 8'b00000000; // Tất cả tắt
            default: LEDR = 8'b00000000;
        endcase
    end
endmodule

module bai_tap_4 (
    input  [3:0] ABCD,
    output F // F=1 nếu thỏa mãn điều kiện
);
    assign F = (ABCD == 4'd0 || ABCD == 4'd6 || ABCD == 4'd12);
endmodule

module bai_tap_5 (
    input  [4:0] ABCDE,
    output is_prime
);
    always @(*) begin
        case (ABCDE)
            5'd2, 5'd3, 5'd5, 5'd7, 5'd11, 5'd13, 
            5'd17, 5'd19, 5'd23, 5'd29, 5'd31: is_prime = 1'b1;
            default: is_prime = 1'b0;
        endcase
    end
endmodule
