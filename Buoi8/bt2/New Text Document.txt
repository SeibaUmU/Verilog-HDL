module bai8(
    input [5:0] SW, 
    output [5:0] LEDR
);
    wire [2:0] a, b;
    wire [2:0] s;
    wire n1, n2, n3, m1, m2, cout; 

    assign a = SW[2:0];
    assign b = SW[5:3];
	 
	 assign LEDR[0] = SW[0] & SW[3];

    cong1bit u0(n1, LEDR[1], SW[0] & SW[4], SW[1] & SW[3], 1'b0);
	 cong1bit u1(n2, s[0], SW[2] & SW[3], SW[1] & SW[4], n1);
	 cong1bit u2(n3, s[1], SW[2] & SW[4], 1'b0, n2);
	 cong1bit u3(m1, LEDR[2], s[0], SW[5] & SW[0], 1'b0);
	 cong1bit u4(m2, LEDR[3], s[1], SW[5] & SW[1], m1);
	 cong1bit u5(cout, LEDR[4], n3, SW[5] & SW[2], m2);

    assign LEDR[5]   = cout; 
endmodule

module cong1bit(output co, output s, input a, input b, input ci);
    wire n1, n2, n3, n4;

    xor(n1, a, b);
    xor(s, ci, n1);
    
    and(n3, a, b);    
    and(n4, ci, n1);
    or(co, n3, n4);
endmodule
