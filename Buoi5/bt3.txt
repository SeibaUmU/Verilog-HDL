module bt3 (
    input SW[0],          // Xung cloSW[0] cạnh lên
    input SW[1],         // Reset không đồng bộ, tích cực mức thấp
    output reg [4:0] LEDR  // Ngõ ra 5 bit (vì 2^4 < 20 < 2^5)
);

    // Khối điều khiển hành vi
    always @(negedge SW[0]) begin
        if (SW[1]) begin
            // Reset không đồng bộ: khi SW[1] = 0, LEDR về 0 ngay lập tức
            LEDR <= 5'b00000;
        end 
        else begin
            // Kiểm tra điều kiện MOD-20 (đến 19 thì reset về 0)
            if (LEDR >= 5'd19) begin
                LEDR <= 5'b00000;
            end 
            else begin
                LEDR <= LEDR + 1'b1;
            end
        end
    end

endmodule
