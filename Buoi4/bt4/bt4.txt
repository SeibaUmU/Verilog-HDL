module mux8_1(S, A, B, C, D, E, M);
input [2:0]S, A, B, C, D, E;
output [2:0]M;
reg [2:0]M;
	always @(*)
		begin
			case (S)
				3'd0: M = A;
				3'd1: M = B;
				3'd2: M = C;
				3'd3: M = D;
				3'd4: M = E;
				default: M = S;
			endcase
		end
endmodule
