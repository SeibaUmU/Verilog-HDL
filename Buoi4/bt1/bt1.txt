module led7seg(in,out);
input [3:0]in;
output [6:0]out;
reg [6:0]out;
always @(in)
begin
case(in)
				4'b0000: out = 7'b1000000; //  00
				4'b0001: out = 7'b1111001; //  01
				4'b0010: out = 7'b0100100; //  02
				4'b0011: out = 7'b0110000; //  03
				4'b0100: out = 7'b0011001; //  04
				4'b0101: out = 7'b0010010; //  05
				4'b0110: out = 7'b0000010; //  06
				4'b0111: out = 7'b1111000; //  07
				4'b1000: out = 7'b0000000; //  08
				4'b1001: out = 7'b0010000; //  09
				default out = 7'b11111111; //off
				endcase
end		
endmodule
