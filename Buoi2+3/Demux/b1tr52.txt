// ==========================================
// Module chính: Demux 1-to-8 (Ghép từ 2 bộ Demux 1-4)
// ==========================================
module demux1_8_board(LEDR, SW);

    // Khai báo theo phong cách giáo trình
    input [3:0] SW;         // SW[0]: Dữ liệu in | SW[3:1]: Chân chọn (Select)
    output [7:0] LEDR;      // 8 ngõ ra

    // Khai báo các đường dây Enable để điều khiển 2 bộ 1-4
    wire en0; 
    wire en1;

    // Phân luồng Enable dựa vào bit chọn cao nhất SW[3]
    // Vì Demux 1-4 tích cực mức CAO nên:
    // - SW[3] = 0 -> Nửa dưới chạy (en0 = 1)
    // - SW[3] = 1 -> Nửa trên chạy (en1 = 1)
    assign en0 = ~SW[3]; 
    assign en1 =  SW[3]; 

    // -----------------------------------------------------------
    // GỌI MODULE CON: Khởi tạo 2 bộ Demux 1-to-4
    // -----------------------------------------------------------
    
    // Nửa dưới (Ngõ ra từ 0 đến 3)
    demux1_4_behavioral demux0 (
        .y(LEDR[3:0]), 
        .in(SW[0]),      // Dữ liệu ngõ vào dùng chung
        .sel(SW[2:1]),   // 2 bit chọn thấp dùng chung
        .en(en0)         // Tín hiệu Enable nửa dưới
    );

    // Nửa trên (Ngõ ra từ 4 đến 7)
    demux1_4_behavioral demux1 (
        .y(LEDR[7:4]), 
        .in(SW[0]),      // Dữ liệu ngõ vào dùng chung
        .sel(SW[2:1]),   // 2 bit chọn thấp dùng chung
        .en(en1)         // Tín hiệu Enable nửa trên
    );

endmodule


// ==========================================
// Module con: Demux 1-to-4 (MÔ HÌNH HÀNH VI)
// Enable: Tích cực mức CAO
// ==========================================
module demux1_4_behavioral(y, in, sel, en);
    
    // Khai báo in/out
    input in;           // 1 bit dữ liệu đầu vào
    input [1:0] sel;    // 2 bit chân chọn
    input en;           // Enable tích cực mức CAO
    
    // Bắt buộc dùng kiểu 'reg' cho ngõ ra trong mô hình hành vi
    output reg [3:0] y; 

    // -----------------------------------------------------------
    // Khối always mô tả hành vi của mạch Demultiplexer
    // -----------------------------------------------------------
    always @(*) begin
        if (en == 1'b1) begin
            // Nếu mạch chạy, gán toàn bộ ngõ ra bằng 0 trước để xóa giá trị cũ
            y = 4'b0000; 
            
            // Sau đó, dựa vào chân select để truyền dữ liệu 'in' ra ĐÚNG vị trí
            case (sel)
                2'b00: y[0] = in; // Truyền dữ liệu ra y[0]
                2'b01: y[1] = in; // Truyền dữ liệu ra y[1]
                2'b10: y[2] = in; // Truyền dữ liệu ra y[2]
                2'b11: y[3] = in; // Truyền dữ liệu ra y[3]
            endcase
        end 
        else begin
            // Nếu mạch tắt (en = 0), chặn dữ liệu, toàn bộ ngõ ra ép về 0
            y = 4'b0000;
        end
    end

endmodule
