// ==========================================
// Module chính (4-to-16): Mô hình hành vi điều khiển Enable
// Ngõ vào Enable: Tích cực mức THẤP
// Ngõ ra: Tích cực mức CAO
// ==========================================
module decoder4_16_board(LEDR, SW);

    // Khai báo theo phong cách giáo trình
    input [4:0] SW;         // SW[3:0]: Dữ liệu | SW[4]: Enable (Tích cực THẤP)
    output [15:0] LEDR;     // 16 ngõ ra (Tích cực CAO)

    // Khai báo kiểu 'reg' cho các tín hiệu Enable nội bộ
    reg en0_n; 
    reg en1_n; 

    // -----------------------------------------------------------
    // MÔ HÌNH HÀNH VI: Tạo tín hiệu phân luồng cho 2 bộ 3-to-8
    // -----------------------------------------------------------
    always @(*) begin
        // Kiểm tra chân Enable tổng (SW[4] tích cực mức THẤP)
        if (SW[4] == 1'b0) begin
            // Mạch được phép chạy, xét tiếp SW[3] để chọn nửa dưới hay nửa trên
            if (SW[3] == 1'b0) begin
                en0_n = 1'b0; // Bật bộ 0-7 (Enable của bộ con cũng tích cực thấp)
                en1_n = 1'b1; // Tắt bộ 8-15
            end 
            else begin
                en0_n = 1'b1; // Tắt bộ 0-7
                en1_n = 1'b0; // Bật bộ 8-15
            end
        end 
        else begin
            // Nếu SW[4] = 1 (Mạch tắt), vô hiệu hóa cả hai bộ
            en0_n = 1'b1;
            en1_n = 1'b1;
        end
    end

    // -----------------------------------------------------------
    // GỌI MODULE CON: Khởi tạo 2 bộ Decoder 3-to-8
    // -----------------------------------------------------------
    
    // Nửa dưới (Ngõ ra từ 0 đến 7)
    decoder3_8_structural dec0 (
        .y(LEDR[7:0]), 
        .d(SW[2:0]), 
        .en_n(en0_n)
    );

    // Nửa trên (Ngõ ra từ 8 đến 15)
    decoder3_8_structural dec1 (
        .y(LEDR[15:8]), 
        .d(SW[2:0]), 
        .en_n(en1_n)
    );

endmodule


// ==========================================
// Module con (3-to-8): MÔ HÌNH CẤU TRÚC
// Ngõ ra: Tích cực mức CAO
// ==========================================
module decoder3_8_structural(y, d, en_n);
    
    // Khai báo in/out
    input [2:0] d;      // 3 bit dữ liệu
    input en_n;         // Enable tích cực mức THẤP
    output [7:0] y;     // 8 bit ngõ ra tích cực mức CAO

    // Các đường dây chứa tín hiệu đảo
    wire nd0, nd1, nd2, en;

    // Tạo tín hiệu đảo bằng cổng NOT
    not u_en (en, en_n);     // Đảo en_n thành en (mức cao) để đưa vào cổng AND
    not u_d0 (nd0, d[0]);
    not u_d1 (nd1, d[1]);
    not u_d2 (nd2, d[2]);

    // -----------------------------------------------------------
    // SỬ DỤNG CỔNG AND: Vì ngõ ra tích cực mức CAO
    // Nếu Enable = 1 và Min-term đúng thì ngõ ra = 1
    // -----------------------------------------------------------
    and a0 (y[0], en, nd2, nd1, nd0); // y[0] = en & ~d2 & ~d1 & ~d0
    and a1 (y[1], en, nd2, nd1, d[0]); // y[1] = en & ~d2 & ~d1 &  d0
    and a2 (y[2], en, nd2, d[1], nd0); // y[2] = en & ~d2 &  d1 & ~d0
    and a3 (y[3], en, nd2, d[1], d[0]); // y[3] = en & ~d2 &  d1 &  d0
    and a4 (y[4], en, d[2], nd1, nd0); // y[4] = en &  d2 & ~d1 & ~d0
    and a5 (y[5], en, d[2], nd1, d[0]); // y[5] = en &  d2 & ~d1 &  d0
    and a6 (y[6], en, d[2], d[1], nd0); // y[6] = en &  d2 &  d1 & ~d0
    and a7 (y[7], en, d[2], d[1], d[0]); // y[7] = en &  d2 &  d1 &  d0

endmodule
