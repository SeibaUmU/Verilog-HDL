// ==========================================
// Module chính: Mạch so sánh 4-bit (Ghép từ 4 bộ 1-bit)
// ==========================================
module comp4bit_board(LEDR, SW);

    // Khai báo theo phong cách giáo trình
    input [8:0] SW;     // SW[3:0]: Số A | SW[7:4]: Số B | SW[8]: Enable
    output [2:0] LEDR;  // LEDR[2]: A>B | LEDR[1]: A=B | LEDR[0]: A<B

    // Khai báo các đường dây chứa kết quả của 4 tầng so sánh
    wire g3, e3, l3; // Kết quả tầng MSB (Bit 3)
    wire g2, e2, l2; // Kết quả tầng 2
    wire g1, e1, l1; // Kết quả tầng 1
    wire g0, e0, l0; // Kết quả tầng LSB (Bit 0)

    // -----------------------------------------------------------
    // GỌI MODULE CON: Khởi tạo 4 bộ so sánh 1-bit
    // -----------------------------------------------------------
    
    // 1. Tầng MSB (Bit 3): A[3] và B[3]
    // Chân Enable nhận trực tiếp từ SW[8]
    comp1bit_dataflow comp_bit3 (
        .g(g3), .e(e3), .l(l3),
        .a(SW[3]), .b(SW[7]), .en(SW[8])
    );

    // 2. Tầng 2: A[2] và B[2]
    // Chân Enable nhận từ ngõ ra E của tầng 3
    comp1bit_dataflow comp_bit2 (
        .g(g2), .e(e2), .l(l2),
        .a(SW[2]), .b(SW[6]), .en(e3)
    );

    // 3. Tầng 1: A[1] và B[1]
    // Chân Enable nhận từ ngõ ra E của tầng 2
    comp1bit_dataflow comp_bit1 (
        .g(g1), .e(e1), .l(l1),
        .a(SW[1]), .b(SW[5]), .en(e2)
    );

    // 4. Tầng LSB (Bit 0): A[0] và B[0]
    // Chân Enable nhận từ ngõ ra E của tầng 1
    comp1bit_dataflow comp_bit0 (
        .g(g0), .e(e0), .l(l0),
        .a(SW[0]), .b(SW[4]), .en(e1)
    );

    // -----------------------------------------------------------
    // TỔNG HỢP KẾT QUẢ ĐẦU RA 
    // Dùng Dataflow (assign) để gộp các tín hiệu lại
    // -----------------------------------------------------------
    
    // A > B: Xảy ra khi bất kỳ tầng nào phát hiện lớn hơn
    assign LEDR[2] = g3 | g2 | g1 | g0;

    // A = B: Xảy ra khi tầng cuối cùng (LSB) báo bằng nhau
    assign LEDR[1] = e0;

    // A < B: Xảy ra khi bất kỳ tầng nào phát hiện nhỏ hơn
    assign LEDR[0] = l3 | l2 | l1 | l0;

endmodule


// ==========================================
// Module con: Mạch so sánh 1-bit (MÔ HÌNH LUỒNG DỮ LIỆU)
// Enable: Tích cực mức CAO
// ==========================================
module comp1bit_dataflow(g, e, l, a, b, en);
    
    // Khai báo in/out
    input a, b;         // 2 bit cần so sánh
    input en;           // Enable tích cực mức CAO
    output g, e, l;     // Ngõ ra kiểu wire mặc định cho Dataflow

    // -----------------------------------------------------------
    // Lệnh assign mô tả dòng chảy dữ liệu bằng biểu thức Boolean
    // Mọi kết quả đều phải AND ( & ) với tín hiệu Enable (en).
    // Nếu en = 0, toàn bộ biểu thức bị kéo về 0.
    // -----------------------------------------------------------
    
    // Greater (a > b): a phải bằng 1 và b phải bằng 0 (~b)
    assign g = en & (a & ~b);

    // Equal (a == b): Dùng phép toán so sánh bằng (==) hoặc cổng XNOR (~(a^b))
    assign e = en & (a == b);

    // Less (a < b): a phải bằng 0 (~a) và b phải bằng 1
    assign l = en & (~a & b);

endmodule
