module bt3(KEY, CLOCK_50, HEX0, HEX1, HEX2, HEX3, HEX4, HEX5, HEX6, HEX7);
input [1:0]KEY;
input CLOCK_50;
output reg [6:0]HEX0;
output [6:0] HEX1, HEX2, HEX3, HEX4, HEX5, HEX6, HEX7; 

wire RS = ~KEY[1];

assign HEX1 = 7'b1111111;
assign HEX2 = 7'b1111111;
assign HEX3 = 7'b1111111;
assign HEX4 = 7'b1111111;
assign HEX5 = 7'b1111111;
assign HEX6 = 7'b1111111;
assign HEX7 = 7'b1111111;


parameter s0 = 3'b000;
parameter s1 = 3'b001;
parameter s2 = 3'b010;
parameter s3 = 3'b011;
parameter s4 = 3'b100;
parameter s5 = 3'b101;

reg [2:0]current_state, next_state;
integer q = 0;
reg clock_2s = 1'b0;

always @ (posedge CLOCK_50)
begin
    q = q + 1;
    if (q == 50000000)
    begin
        clock_2s = ~clock_2s;
        q = 0;
    end
end

always @ (*)
begin
    case (current_state)
        s0: next_state = s1;
        s1: next_state = s2;
        s2: next_state = s3;
        s3: next_state = s4;
        s4: next_state = s5;
        s5: next_state = s0;
        default next_state = s0;
    endcase
end

always @ (posedge clock_2s or posedge RS)
begin
    if (RS) current_state <= s0;
    else current_state <= next_state;
end

always @ (*)
begin
    if (RS)
        HEX0 = 7'b1111111;
    else
        case (current_state)
            s0: HEX0 = 7'b0001000;
            s1: HEX0 = 7'b0001100;
            s2: HEX0 = 7'b1111111;
            s3: HEX0 = 7'b0001100;
            s4: HEX0 = 7'b1000111;
            s5: HEX0 = 7'b0000110;
            default: HEX0 = 7'b1111111;
        endcase
end

endmodule
